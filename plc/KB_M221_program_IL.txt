(* KALBAR PDAM - PANEL MONITORING KUALITAS AIR (pH, KLORIN, TURBIDITY) *)
(* PLC Schneider TM221CE16R + TM3AI4 - EcoStruxure Machine Expert Basic *)
(* Program IL reversible: ketik/tempel per rung dalam mode IL, lalu ubah tampilan ke Ladder *)
(* Ref: ProMinent BAMa, DULCOMETER D1Cb x3, DULCO turb C TUC 5 - lihat README *)


(* ================= POU0  Inisialisasi & perintah ================= *)
(* Rung 001: Default parameter (sekali saat memori kosong) *)
(*   Bila %MW60 belum 12345 (PLC baru / memori hilang) semua setpoint diisi nilai contoh. Setelah itu nilai hanya diubah dari HMI. *)
    LD [%MW60 <> 12345]
    [%MW0 := 60]
    [%MW1 := 10]
    [%MW2 := 60]
    [%MW3 := 650]
    [%MW4 := 850]
    [%MW5 := 20]
    [%MW6 := 100]
    [%MW7 := 300]
    [%MW8 := 0]
    [%MW9 := 0]
    [%MW10 := 30]
    [%MW11 := 100]
    [%MW60 := 12345]

(* Rung 002: Muat preset timer dari parameter *)
(*   Preset timer diambil dari %MW parameter tiap scan. Timer dengan preset tetap diisi di konfigurasi. *)
    [%TM0.P := %MW10]
    [%TM1.P := %MW0]
    [%TM2.P := %MW1]
    [%TM3.P := %MW2]
    [%TM4.P := %MW2]
    [%TM5.P := %MW2]
    [%TM6.P := %MW2]
    [%TM7.P := %MW2]

(* Rung 003: Tunda start-up *)
(*   Setelah PLC RUN, alarm kualitas menunggu P_T_STARTUP agar sinyal analog dan instrumen stabil. *)
    BLK %TM0
    LD [%MW60 = 12345]
    IN
    OUT_BLK
    LD Q
    ST %M1
    END_BLK

(* Rung 004: Perintah ACK (tombol panel tepi naik / HMI) *)
    LD %I0.6
    ANDN %M8
    OR %M300
    ST %M7

(* Rung 005: Memori tepi tombol reset *)
    LD %I0.6
    ST %M8

(* Rung 006: Heartbeat untuk HMI/SCADA *)
    LD %S6
    ST %M9


(* ================= POU1  Nilai analog ================= *)
(* Rung 007: Salin nilai analog *)
    [%MW100 := %IW1.0]
    [%MW101 := %IW1.1]
    [%MW102 := %IW1.2]
    [%MW104 := %IW1.3]

(* Rung 008: Selisih pH #1 - pH #2 *)
    [%MW106 := %MW100 - %MW101]

(* Rung 009:   nilai absolut *)
    LD [%MW106 < 0]
    [%MW106 := 0 - %MW106]

(* Rung 010: Turbidity dari 4-20 mA *)
    LD [%MW9 = 0]
    [%MW103 := %MW104]

(* Rung 011: Turbidity dari Modbus *)
(*   TUC 5 hanya bisa memilih SATU output: 4-20 mA atau RS-485 (menu output). Samakan dengan P_TU_SRC. *)
    LD [%MW9 = 1]
    [%MW103 := %MW105]

(* Rung 012: Gangguan sinyal pH #1 *)
(*   Status kanal TM3 bukan 0 = under/over range atau kabel putus (D1Cb fault = 3,6 mA). *)
    LD [%IWS1.0 <> 0]
    ST %M214

(* Rung 013: Gangguan sinyal pH #2 *)
    LD [%IWS1.1 <> 0]
    ST %M215

(* Rung 014: Gangguan sinyal klorin *)
    LD [%IWS1.2 <> 0]
    ST %M216

(* Rung 015: Gangguan sinyal turbidity *)
(*   Set ERLV TUC 5 = 2 mA: fault alat muncul sebagai under-range. *)
    LD [%MW9 = 0]
    ANDN %M313
    AND [%IWS1.3 <> 0]
    ST %M217

(* Rung 016: Cermin status I/O fisik untuk HMI *)
    [%MW120 := %I0.0:9]
    [%MW121 := %Q0.0:7]

(* Rung 017:   cermin bit FS_SAMPLE *)
(*   Bit cermin dibaca SCADA sebagai coil (%M40-46 = DI). *)
    LD %I0.0
    ST %M40

(* Rung 018:   cermin bit PH1_OK *)
    LD %I0.1
    ST %M41

(* Rung 019:   cermin bit PH2_OK *)
    LD %I0.2
    ST %M42

(* Rung 020:   cermin bit CL_OK *)
    LD %I0.3
    ST %M43

(* Rung 021:   cermin bit TU_A1_OK *)
    LD %I0.4
    ST %M44

(* Rung 022:   cermin bit TU_A2_OK *)
    LD %I0.5
    ST %M45

(* Rung 023:   cermin bit PB_RESET *)
    LD %I0.6
    ST %M46


(* ================= POU2  Aliran sampel & PAUSE dosing ================= *)
(* Rung 024: Aliran sampel stabil *)
    BLK %TM1
    LD %I0.0
    IN
    OUT_BLK
    LD Q
    ST %M2
    END_BLK

(* Rung 025: Alarm aliran sampel hilang *)
    BLK %TM2
    LDN %I0.0
    AND %M1
    IN
    OUT_BLK
    LD Q
    ST %M200
    END_BLK

(* Rung 026: PAUSE dosing pH #1 *)
(*   Relai ON = pause. Set input pause D1Cb ke 'aktif saat kontak tertutup'. Bila PLC mati, dosing tetap jalan (tidak berhenti mendadak). *)
    LDN %M2
    OR %M310
    ST %Q0.0

(* Rung 027: Nilai pH #1 valid *)
    LD %M2
    AND %M1
    ANDN %M214
    ANDN %M310
    ST %M3

(* Rung 028: PAUSE dosing pH #2 *)
    LDN %M2
    OR %M311
    ST %Q0.1

(* Rung 029: Nilai pH #2 valid *)
    LD %M2
    AND %M1
    ANDN %M215
    ANDN %M311
    ST %M4

(* Rung 030: PAUSE dosing klorin *)
    LDN %M2
    OR %M312
    ST %Q0.2

(* Rung 031: Nilai klorin valid *)
    LD %M2
    AND %M1
    ANDN %M216
    ANDN %M312
    ST %M5

(* Rung 032: Nilai turbidity valid *)
(*   TUC 5 punya flow cell sendiri, tidak tergantung aliran BAMa. *)
    LD %M1
    ANDN %M313
    AND( [%MW9 = 0]
    ANDN %M217
    OR( [%MW9 = 1]
    ANDN %M218
    ANDN %M219
    )
    )
    ST %M6


(* ================= POU3  Alarm kualitas air ================= *)
(* Rung 033: pH #1 di luar batas *)
    LD %M3
    AND( [%MW100 < %MW3]
    OR [%MW100 > %MW4]
    )
    ST %M13

(* Rung 034:   tunda alarm pH #1 *)
    BLK %TM3
    LD %M13
    IN
    END_BLK

(* Rung 035:   pH #1 rendah *)
    LD %TM3.Q
    AND [%MW100 < %MW3]
    ST %M206

(* Rung 036:   pH #1 tinggi *)
    LD %TM3.Q
    AND [%MW100 > %MW4]
    ST %M207

(* Rung 037: pH #2 di luar batas *)
    LD %M4
    AND( [%MW101 < %MW3]
    OR [%MW101 > %MW4]
    )
    ST %M14

(* Rung 038:   tunda alarm pH #2 *)
    BLK %TM4
    LD %M14
    IN
    END_BLK

(* Rung 039:   pH #2 rendah *)
    LD %TM4.Q
    AND [%MW101 < %MW3]
    ST %M208

(* Rung 040:   pH #2 tinggi *)
    LD %TM4.Q
    AND [%MW101 > %MW4]
    ST %M209

(* Rung 041: klorin di luar batas *)
    LD %M5
    AND( [%MW102 < %MW5]
    OR [%MW102 > %MW6]
    )
    ST %M15

(* Rung 042:   tunda alarm klorin *)
    BLK %TM5
    LD %M15
    IN
    END_BLK

(* Rung 043:   klorin rendah *)
    LD %TM5.Q
    AND [%MW102 < %MW5]
    ST %M210

(* Rung 044:   klorin tinggi *)
    LD %TM5.Q
    AND [%MW102 > %MW6]
    ST %M211

(* Rung 045: Turbidity di atas batas *)
    LD %M6
    AND [%MW103 > %MW7]
    ST %M16

(* Rung 046:   tunda alarm turbidity *)
    BLK %TM6
    LD %M16
    IN
    OUT_BLK
    LD Q
    ST %M212
    END_BLK

(* Rung 047: Selisih pH di atas batas *)
(*   Aktifkan (P_PH_DIFF > 0) hanya bila kedua elektroda mengukur air yang sama. *)
    LD [%MW8 > 0]
    AND %M3
    AND %M4
    AND [%MW106 > %MW8]
    ST %M17

(* Rung 048:   tunda alarm selisih pH *)
    BLK %TM7
    LD %M17
    IN
    OUT_BLK
    LD Q
    ST %M213
    END_BLK

(* Rung 049: Zona pH #1: reset *)
(*   Zona untuk indikasi bar SCADA (langsung, tanpa tunda): 1 rendah, 2 menengah/normal, 3 tinggi. *)
    [%MW130 := 0]

(* Rung 050:   zona pH #1 menengah *)
    LD %M3
    [%MW130 := 2]

(* Rung 051:   zona pH #1 rendah *)
    LD %M3
    AND [%MW100 < %MW3]
    [%MW130 := 1]

(* Rung 052:   zona pH #1 tinggi *)
    LD %M3
    AND [%MW100 > %MW4]
    [%MW130 := 3]

(* Rung 053: Zona pH #2: reset *)
    [%MW131 := 0]

(* Rung 054:   zona pH #2 menengah *)
    LD %M4
    [%MW131 := 2]

(* Rung 055:   zona pH #2 rendah *)
    LD %M4
    AND [%MW101 < %MW3]
    [%MW131 := 1]

(* Rung 056:   zona pH #2 tinggi *)
    LD %M4
    AND [%MW101 > %MW4]
    [%MW131 := 3]

(* Rung 057: Zona klorin: reset *)
    [%MW132 := 0]

(* Rung 058:   zona klorin menengah *)
    LD %M5
    [%MW132 := 2]

(* Rung 059:   zona klorin rendah *)
    LD %M5
    AND [%MW102 < %MW5]
    [%MW132 := 1]

(* Rung 060:   zona klorin tinggi *)
    LD %M5
    AND [%MW102 > %MW6]
    [%MW132 := 3]

(* Rung 061: Zona turbidity: reset *)
    [%MW133 := 0]

(* Rung 062:   zona turbidity rendah (baik) *)
    LD %M6
    [%MW133 := 1]

(* Rung 063:   zona turbidity menengah *)
    LD %M6
    AND [%MW103 > %MW11]
    [%MW133 := 2]

(* Rung 064:   zona turbidity tinggi *)
    LD %M6
    AND [%MW103 > %MW7]
    [%MW133 := 3]

(* Rung 065: Kualitas air normal *)
    LD %M3
    AND %M4
    AND %M5
    AND %M6
    ANDN %M13
    ANDN %M14
    ANDN %M15
    ANDN %M16
    ST %M12


(* ================= POU4  Alarm dari instrumen ================= *)
(* Rung 066: D1Cb pH #1 alarm *)
    LDN %I0.1
    AND %M1
    ST %M201

(* Rung 067: D1Cb pH #2 alarm *)
    LDN %I0.2
    AND %M1
    ST %M202

(* Rung 068: D1Cb klorin alarm *)
    LDN %I0.3
    AND %M1
    ST %M203

(* Rung 069: TUC 5 alarm 1 (turbidity tinggi) *)
    LDN %I0.4
    AND %M1
    ANDN %M313
    ST %M204

(* Rung 070: TUC 5 alarm 2 (fault) *)
    LDN %I0.5
    AND %M1
    ANDN %M313
    ST %M205


(* ================= POU5  Modbus RTU SL1 - TUC 5 (opsional) ================= *)
(* Rung 071: Pulsa polling 1 s *)
    BLK %TM8
    LD [%MW9 = 1]
    ANDN %M18
    IN
    OUT_BLK
    LD Q
    ST %M18
    END_BLK

(* Rung 072: Eksekusi READ_VAR TUC 5 *)
    BLK %READ_VAR0
    LD [%MW109 = 0]
    AND %M18
    EXEC
    END_BLK

(* Rung 073:   mulai tunggu jawaban *)
    LD [%MW109 = 0]
    AND %M18
    [%MW109 := 1]

(* Rung 074:   selesai OK: ambil data *)
(*   Bila nilai tidak masuk akal (mis. 1E-38), word order terbalik: tukar %MW300/%MW301 sebelum konversi. *)
    LD [%MW109 = 1]
    AND %READ_VAR0.D
    [%MF320 := %MF300 * 100.0]
    [%MW105 := REAL_TO_INT(%MF320)]
    [%MW107 := %MW309]
    [%MW108 := %MW311]
    [%MW110 := 0]
    [%MW109 := 0]

(* Rung 075:   error komunikasi *)
    LD [%MW109 = 1]
    AND %READ_VAR0.E
    [%MW110 := %MW110 + 1]
    [%MW109 := 0]

(* Rung 076: Batasi langkah polling *)
    LD [%MW109 > 1]
    OR [%MW9 <> 1]
    [%MW109 := 0]

(* Rung 077: Batasi counter error *)
    LD [%MW110 > 100]
    [%MW110 := 100]

(* Rung 078: Komunikasi TUC 5 putus *)
    LD [%MW9 = 1]
    ANDN %M313
    AND [%MW110 >= 3]
    ST %M218

(* Rung 079: TUC 5 melaporkan error *)
(*   Reg 30010 = 6 (error) atau bit error reg 30012 (desiccant, lampu, ultrasonik, kalibrasi). *)
    LD [%MW9 = 1]
    ANDN %M313
    ANDN %M218
    AND( [%MW107 = 6]
    OR [%MW108 <> 0]
    )
    ST %M219


(* ================= POU6  Alarm, buzzer & lampu ================= *)
(* Rung 080: Pak bit alarm ke word *)
    [%MW200 := %M200:16]
    [%MW201 := %M216:16]

(* Rung 081: Deteksi alarm baru word 0 *)
    [%MW208 := NOT %MW204]
    [%MW208 := %MW200 AND %MW208]

(* Rung 082:   alarm baru word 0 -> buzzer *)
    LD [%MW208 <> 0]
    S %M11

(* Rung 083: Deteksi alarm baru word 1 *)
    [%MW208 := NOT %MW205]
    [%MW208 := %MW201 AND %MW208]

(* Rung 084:   alarm baru word 1 -> buzzer *)
    LD [%MW208 <> 0]
    S %M11

(* Rung 085: Simpan word alarm *)
    [%MW204 := %MW200]
    [%MW205 := %MW201]

(* Rung 086: Ada alarm aktif *)
    LD [%MW200 <> 0]
    OR [%MW201 <> 0]
    ST %M10

(* Rung 087: ACK: matikan buzzer *)
    LD %M7
    R %M11

(* Rung 088: Buzzer *)
    LD %M11
    ST %Q0.4

(* Rung 089: Lampu alarm (kedip bila belum ACK) *)
    LD %M11
    AND %S6
    OR( %M10
    ANDN %M11
    )
    ST %Q0.3

(* Rung 090: Ada dosing di-PAUSE *)
    LDN %M2
    OR %M310
    OR %M311
    OR %M312
    ST %M19

(* Rung 091: Lampu hijau (tetap = normal, kedip = pause) *)
(*   CE16R hanya 7 relay, jadi lampu kuning digabung: lampu hijau berkedip saat dosing di-PAUSE. *)
    LD %M19
    AND %S6
    OR( %M12
    ANDN %M19
    ANDN %M10
    )
    ST %Q0.5

(* Rung 092: Alarm umum ke remote (fail-safe) *)
(*   Relai ON = sehat. PLC mati / kabel putus terbaca sebagai alarm di SCADA. *)
    LDN %M10
    ST %Q0.6

(* Rung 093:   cermin bit Q_PAUSE_PH1 *)
(*   Cermin output (%M50-56 = DO) setelah semua output ditulis. *)
    LD %Q0.0
    ST %M50

(* Rung 094:   cermin bit Q_PAUSE_PH2 *)
    LD %Q0.1
    ST %M51

(* Rung 095:   cermin bit Q_PAUSE_CL *)
    LD %Q0.2
    ST %M52

(* Rung 096:   cermin bit Q_HL_ALM *)
    LD %Q0.3
    ST %M53

(* Rung 097:   cermin bit Q_BUZZER *)
    LD %Q0.4
    ST %M54

(* Rung 098:   cermin bit Q_HL_OK *)
    LD %Q0.5
    ST %M55

(* Rung 099:   cermin bit Q_ALM_REMOTE *)
    LD %Q0.6
    ST %M56


(* ================= POU7  Reset tombol HMI ================= *)
(* Rung 100: Reset HMI_ACK *)
    LD %M300
    R %M300
